/*
 * AutoWorkshop — Arduino PN532 NFC Writer for NTAG215 Cards
 * 
 * Writes the Client App Portal URL directly to an NTAG215 card.
 * Formats the card as NDEF URI (https://) so smartphones (Android/iOS)
 * automatically open the web URL when tapped.
 * Compatible with PN532 module (I2C) and Adafruit_PN532 library.
 */

#include <Wire.h>
#include <Adafruit_PN532.h>

#define PN532_IRQ   (2)
#define PN532_RESET (3)

Adafruit_PN532 nfc(PN532_IRQ, PN532_RESET);

// Target URL to write (without https:// prefix, as https:// is encoded as prefix code 0x04)
const char targetDomain[] = "gearshift2.vercel.app/portal/empresaabcsa";

void setup(void) {
  Serial.begin(115200);
  while (!Serial) delay(10);

  Serial.println("Initializing PN532 for NTAG215...");
  nfc.begin();

  uint32_t versiondata = nfc.getFirmwareVersion();
  if (!versiondata) {
    Serial.println("PN532 module not found.");
    while (1);
  }
  nfc.SAMConfig();
  Serial.println("Place NTAG215 card to write Client App link...");
}

void loop(void) {
  uint8_t success;
  uint8_t uid[] = { 0, 0, 0, 0, 0, 0, 0 };  
  uint8_t uidLength;
  
  success = nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, uid, &uidLength);
  
  if (success) {
    Serial.println("NTAG215 Card Found! Writing NDEF URI Portal URL...");

    // Step 1: Initialize NTAG215 Capability Container (Page 3) for NDEF compatibility
    uint8_t ccPage[4] = { 0xE1, 0x10, 0x6D, 0x00 };
    nfc.ntag2xx_WritePage(3, ccPage);

    // Step 2: Build NDEF URI Record (https://)
    uint8_t domainLen = strlen(targetDomain);
    uint8_t recordPayloadLen = 1 + domainLen; // 0x04 (https://) + domain string
    uint8_t ndefMsgLen = 4 + recordPayloadLen; // 0xD1, 0x01, recordPayloadLen, 0x55 ('U')

    uint8_t payloadBytes[128];
    uint16_t idx = 0;

    payloadBytes[idx++] = 0x03;             // NDEF TLV Tag
    payloadBytes[idx++] = ndefMsgLen;       // TLV Length
    payloadBytes[idx++] = 0xD1;             // NDEF Header (MB=1, ME=1, SR=1, TNF=1)
    payloadBytes[idx++] = 0x01;             // Record Type Length ('U')
    payloadBytes[idx++] = recordPayloadLen; // Record Payload Length
    payloadBytes[idx++] = 0x55;             // Record Type 'U' (URI)
    payloadBytes[idx++] = 0x04;             // URI Prefix Code (0x04 = https://)

    for (uint8_t i = 0; i < domainLen; i++) {
      payloadBytes[idx++] = targetDomain[i];
    }
    payloadBytes[idx++] = 0xFE;             // Terminator TLV Tag

    // Step 3: Write payload starting at Page 4
    uint8_t totalPages = (idx + 3) / 4;
    uint8_t pageBuffer[4];

    for (uint8_t page = 0; page < totalPages; page++) {
      for (uint8_t b = 0; b < 4; b++) {
        uint16_t byteIdx = page * 4 + b;
        pageBuffer[b] = (byteIdx < idx) ? payloadBytes[byteIdx] : 0x00;
      }
      nfc.ntag2xx_WritePage(4 + page, pageBuffer);
    }

    Serial.println("SUCCESS! NDEF Web Link written onto NFC Card.");
    Serial.print("URL: https://");
    Serial.println(targetDomain);
    
    while(1) { delay(1000); }
  }
  delay(2000);
}
